// STEPPER MOTOR INTERFACE

#include <htc.h>
#define _XTAL_FREQ 20000000

void main()
{
    int i;
    int s[] = {0x11, 0x22, 0x44, 0x88};

    TRISB = 0x00;
    PORTB = 0x00;

    // Forward rotation
    for (int j = 0; j <= 200; j++)
    {
        for (i = 0; i <= 3; i++)
        {
            PORTB = s[i];
            __delay_ms(5);
        }
    }
    __delay_ms(2000);

    // Reverse rotation
    for (int j = 0; j <= 200; j++)
    {
        for (i = 3; i >= 0; i--)
        {
            PORTB = s[i];
            __delay_ms(5);
        }
    }
    __delay_ms(2000);
}
