#include <htc.h>
#define _XTAL_FREQ 20000000

unsigned char seg[10] = {0x3F,0x06,0x5B,0x4F,0x66,0x6D,0x7D,0x07,0x7F,0x6F};
unsigned int count = 0;
unsigned char i = 1;          // start at 1, because 0 is already shown

void interrupt ISR()
{
    if(T0IF)
    {
        TMR0 = 61;
        T0IF = 0;
        count++;
        if(count >= 100)
        {
            count = 0;
            PORTB = seg[i];
            i++;
            if(i >= 10) i = 0;
        }
    }
}

void main()
{
    TRISB = 0;
    PORTB = seg[0];
    OPTION_REG = 0x07;
    TMR0 = 61;
    T0IE = 1;
    GIE = 1;
    while(1)
    {
    }
}
